#include <Arduino.h>

// ================= IR SENSOR PINS =================
#define IR1 34  // Leftmost
#define IR2 35
#define IR3 32  // Center
#define IR4 33
#define IR5 25  // Rightmost

// ================= MOTOR DRIVER PINS (L298N style) =================
#define ENA 26  // Left PWM
#define IN1 27  // Left forward
#define IN2 14  // Left backward

#define ENB 12  // Right PWM
#define IN3 13  // Right forward
#define IN4 15  // Right backward

// ================= PWM SETTINGS (ESP32) =================
#define CH_LEFT  0
#define CH_RIGHT 1
#define PWM_FREQ 1000
#define PWM_RES  8

// ================= PID PARAMETERS =================
float Kp = 28.0;  // Proportional gain
float Ki = 0.0;   // Integral gain (disabled to avoid windup)
float Kd = 16.0;  // Derivative gain

float error = 0, lastError = 0;
float integral = 0, derivative = 0, correction = 0;

// ================= SPEED SETTINGS =================
int baseSpeed = 150;
int maxSpeed = 200;
int minSpeed = 0;

int sharpTurnSpeed = 230;
int searchSpeed = 150;
int slowForward = 100;

// ================= SENSOR VALUES (0=black line, 1=white) =================
int s1, s2, s3, s4, s5;

// ================= MEMORY & TIMING =================
int lastDirection = 0;       // -1=left, 1=right, 0=center
int sharpTurnMemory = 0;     // For continuing turns
unsigned long sharpTurnTimeout = 1600;  // 1.6 seconds

// ================= SETUP =================
void setup() {
  Serial.begin(115200);
  Serial.println("Line Follower Robot Starting...");

  // IR Sensors as INPUT
  pinMode(IR1, INPUT);
  pinMode(IR2, INPUT);
  pinMode(IR3, INPUT);
  pinMode(IR4, INPUT);
  pinMode(IR5, INPUT);

  // Motor pins as OUTPUT
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  // PWM Setup for ESP32
  ledcSetup(CH_LEFT, PWM_FREQ, PWM_RES);
  ledcSetup(CH_RIGHT, PWM_FREQ, PWM_RES);
  ledcAttachPin(ENA, CH_LEFT);
  ledcAttachPin(ENB, CH_RIGHT);

  stopMotors();
  delay(1000);
  Serial.println("Ready to track line!");
}

// ================= MAIN LOOP =================
void loop() {
  readSensors();

  // PRIORITY 1: SHARP TURN DETECTION (s1,s5 override PID)
  if (s1 == 0 && s5 == 1) {  // 0 1 = Sharp LEFT turn
    sharpTurnMemory = -1;
    executeSharpTurn();
    return;
  }
  if (s1 == 1 && s5 == 0) {  // 1 0 = Sharp RIGHT turn
    sharpTurnMemory = 1;
    executeSharpTurn();
    return;
  }
  if (s1 == 0 && s5 == 0) {  // 0 0 = Lost outer sensors, continue last turn
    if (lastDirection < 0) sharpTurnMemory = -1;
    else if (lastDirection > 0) sharpTurnMemory = 1;
    
    if (sharpTurnMemory != 0) {
      executeSharpTurn();
      return;
    }
  }
  sharpTurnMemory = 0;  // Reset memory

  // PRIORITY 2: NORMAL PID TRACKING (s2,s3,s4)
  normalPIDTracking();
}

// ================= READ ALL SENSORS =================
void readSensors() {
  s1 = digitalRead(IR1);
  s2 = digitalRead(IR2);
  s3 = digitalRead(IR3);
  s4 = digitalRead(IR4);
  s5 = digitalRead(IR5);
  
  // Debug print (optional, comment out for speed)
  Serial.printf("Sensors: %d %d %d %d %d\n", s1, s2, s3, s4, s5);
}

// ================= PID POSITION CALCULATION =================
void normalPIDTracking() {
  // Position error based on s2,s3,s4 patterns (s1,s5 ignored here)
  if (s2 == 1 && s3 == 0 && s4 == 1) {
    error = 0;
    lastDirection = 0;
  }
  else if (s2 == 0 && s3 == 0 && s4 == 1) {
    error = -1;
    lastDirection = -1;
  }
  else if (s2 == 0 && s3 == 1 && s4 == 1) {
    error = -2;
    lastDirection = -1;
  }
  else if (s2 == 1 && s3 == 0 && s4 == 0) {
    error = 1;
    lastDirection = 1;
  }
  else if (s2 == 1 && s3 == 1 && s4 == 0) {
    error = 2;
    lastDirection = 1;
  }
  else if (s2 == 0 && s3 == 0 && s4 == 0) {
    moveForward(slowForward, slowForward);
    return;
  }
  else {  // All on (111) or unmatched → lost line
    recoverLostLine();
    return;
  }

  // PID Calculation
  integral += error;
  integral = constrain(integral, -40, 40);  // Anti-windup

  derivative = error - lastError;
  correction = Kp * error + Ki * integral + Kd * derivative;
  lastError = error;

  // Motor speeds
  int leftSpeed  = baseSpeed + (int)correction;
  int rightSpeed = baseSpeed - (int)correction;

  leftSpeed  = constrain(leftSpeed, minSpeed, maxSpeed);
  rightSpeed = constrain(rightSpeed, minSpeed, maxSpeed);

  moveForward(leftSpeed, rightSpeed);
  
  // Debug PID
  Serial.printf("Error: %.1f, Corr: %.1f, L:%d R:%d\n", error, correction, leftSpeed, rightSpeed);
}

// ================= SHARP 180° TURN EXECUTION =================
void executeSharpTurn() {
  unsigned long startTime = millis();

  while (true) {
    readSensors();

    // Execute pivot based on direction
    if (sharpTurnMemory < 0) {  // Left turn (CCW)
      leftForward(sharpTurnSpeed);
      rightBackward(sharpTurnSpeed);
    } else if (sharpTurnMemory > 0) {  // Right turn (CW)
      leftBackward(sharpTurnSpeed);
      rightForward(sharpTurnSpeed);
    }

    // Exit conditions
    if (s3 == 0 && s1 == 1 && s5 == 1) {  // Center on line, outers off track
      stopMotors();
      sharpTurnMemory = 0;
      Serial.println("Sharp turn complete!");
      return;
    }

    if (millis() - startTime > sharpTurnTimeout) {
      stopMotors();
      sharpTurnMemory = 0;
      Serial.println("Sharp turn timeout!");
      return;
    }
  }
}

// ================= LOST LINE RECOVERY =================
void recoverLostLine() {
  if (lastDirection < 0) {  // Last known left deviation
    leftBackward(searchSpeed);
    rightForward(searchSpeed);
  }
  else if (lastDirection > 0) {  // Last known right deviation
    leftForward(searchSpeed);
    rightBackward(searchSpeed);
  }
  else {  // No direction memory
    moveForward(90, 90);
  }
}

// ================= MOTOR CONTROL FUNCTIONS =================
void moveForward(int leftSpeed, int rightSpeed) {
  leftForward(leftSpeed);
  rightForward(rightSpeed);
}

void stopMotors() {
  ledcWrite(CH_LEFT, 0);
  ledcWrite(CH_RIGHT, 0);
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}

void leftForward(int speedVal) {
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  ledcWrite(CH_LEFT, speedVal);
}

void leftBackward(int speedVal) {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);
  ledcWrite(CH_LEFT, speedVal);
}

void rightForward(int speedVal) {
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
  ledcWrite(CH_RIGHT, speedVal);
}

void rightBackward(int speedVal) {
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
  ledcWrite(CH_RIGHT, speedVal);
}
