#include <Arduino.h>
#include <Wire.h>
#include "Adafruit_TCS34725.h"

// ================= IR SENSOR PINS =================
#define IR1 34
#define IR2 35
#define IR3 32
#define IR4 33
#define IR5 25

// ================= MOTOR DRIVER =================
#define ENA 26
#define IN1 27
#define IN2 14
#define ENB 12
#define IN3 13
#define IN4 15

// ================= ULTRASONIC =================
#define TRIG_PIN 4
#define ECHO_PIN 18

// ================= LED =================
#define BULB_PIN 2
#define STATUS_LED 5

// ================= PWM =================
#define CH_LEFT  0
#define CH_RIGHT 1

// ================= COLOR SENSOR =================
Adafruit_TCS34725 tcs = Adafruit_TCS34725(
  TCS34725_INTEGRATIONTIME_50MS,
  TCS34725_GAIN_4X
);

// ================= PID =================
float Kp = 28.0;
float Kd = 18.0;   // 🔥 increased for stability
float lastError = 0;

// ================= SPEED =================
int baseSpeed = 150;
int maxSpeed = 200;
int minSpeed = 0;
int sharpTurnSpeed = 230;  // 🔥 stronger turns
int searchSpeed = 160;

// ================= SENSOR =================
int s1, s2, s3, s4, s5;
int sensorThreshold = 2000;

// ================= MEMORY =================
int lastDirection = 0;
int sharpTurnMemory = 0;
unsigned long sharpTurnTimeout = 2000;

// ================= SETUP =================
void setup() {
  Serial.begin(115200);

  pinMode(IR1, INPUT);
  pinMode(IR2, INPUT);
  pinMode(IR3, INPUT);
  pinMode(IR4, INPUT);
  pinMode(IR5, INPUT);

  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);

  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  pinMode(BULB_PIN, OUTPUT);
  pinMode(STATUS_LED, OUTPUT);

  ledcSetup(CH_LEFT, 1000, 8);
  ledcSetup(CH_RIGHT, 1000, 8);
  ledcAttachPin(ENA, CH_LEFT);
  ledcAttachPin(ENB, CH_RIGHT);

  if (!tcs.begin()) {
    Serial.println("Color sensor not found");
  }
}

// ================= LOOP =================
void loop() {

  float distance = getDistance();

  // ===== COLOR =====
  uint16_t r, g, b, c;
  tcs.getRawData(&r, &g, &b, c);

  bool isRed = (r > g && r > b && r > 60);
  bool isGreen = (g > r && g > b && g > 60);

  readSensors();

  // ================= 🟢 GREEN =================
  if (isGreen && distance > 0 && distance <= 12) {
    stopMotors();

    for (int i = 0; i < 10; i++) {
      digitalWrite(BULB_PIN, HIGH); delay(100);
      digitalWrite(BULB_PIN, LOW);  delay(100);
    }
    return;
  }

  // ================= 🔴 RED =================
  if (isRed && distance > 0 && distance <= 12) {
    stopMotors();
    delay(100);

    // Deep Box avoidance maneuver (From Code 2)
    leftBackward(150); rightBackward(150); delay(500); // Back up
    leftForward(160); rightBackward(160); delay(500);  // Turn Right
    moveForward(160, 160); delay(800);                 // Move forward past obstacle
    leftBackward(150); rightForward(150); delay(500);  // Turn Left
    moveForward(160, 160); delay(1000);                // Move forward to cross line
    leftBackward(150); rightForward(150); delay(500);  // Turn Left to face line direction

    unsigned long start = millis();
    while (true) {
      readSensors();
      leftForward(160); rightForward(120); // Sweep curve
      if (s3 == 0 || s2 == 0 || s4 == 0) break; // Found line
      if (millis() - start > 3000) break;       // Timeout
    }
    return;
  }

  // ================= 🔥 ALL IR 11111 =================
  if (s1 == 1 && s2 == 1 && s3 == 1 && s4 == 1 && s5 == 1) {
    stopMotors();
    digitalWrite(STATUS_LED, HIGH);
    
    // LOST LINE → use memory to find it again
    if (lastDirection != 0) {
      sharpTurnMemory = lastDirection;
      executeSharpTurn();
    }
    return;
  } else {
    digitalWrite(STATUS_LED, LOW);
  }

  // ================= 🔥 HARD TURN LOGIC =================
  // EXTREME LEFT
  if (s1 == 0 && s2 == 0 && s3 == 1 && s4 == 1 && s5 == 1) {
    sharpTurnMemory = -1;
    executeSharpTurn();
    return;
  }

  // EXTREME RIGHT
  if (s5 == 0 && s4 == 0 && s3 == 1 && s2 == 1 && s1 == 1) {
    sharpTurnMemory = 1;
    executeSharpTurn();
    return;
  }

  // ================= NORMAL PID =================
  normalPIDTracking();
}

// ================= PID =================
void normalPIDTracking() {

  float error = 0;

  if      (s3 == 0) { error = 0; lastDirection = 0; }
  else if (s2 == 0) { error = -1; lastDirection = -1; }
  else if (s4 == 0) { error = 1; lastDirection = 1; }
  else if (s1 == 0) { error = -2; lastDirection = -1; }
  else if (s5 == 0) { error = 2; lastDirection = 1; }
  else { recoverLostLine(); return; }

  float derivative = error - lastError;
  float correction = (Kp * error) + (Kd * derivative);
  lastError = error;

  int leftSpeed  = constrain(baseSpeed + correction, minSpeed, maxSpeed);
  int rightSpeed = constrain(baseSpeed - correction, minSpeed, maxSpeed);

  moveForward(leftSpeed, rightSpeed);
}

// ================= SHARP TURN =================
void executeSharpTurn() {
  unsigned long start = millis();

  while (millis() - start < sharpTurnTimeout) {
    readSensors();

    if (sharpTurnMemory < 0) {
      leftForward(sharpTurnSpeed);
      rightBackward(sharpTurnSpeed);
    } else {
      leftBackward(sharpTurnSpeed);
      rightForward(sharpTurnSpeed);
    }

    if (s3 == 0) break; // Stop turning when center sensor finds line
  }

  stopMotors();
}

// ================= RECOVERY =================
void recoverLostLine() {
  if (lastDirection < 0) {
    leftBackward(searchSpeed);
    rightForward(searchSpeed);
  } else {
    leftForward(searchSpeed);
    rightBackward(searchSpeed);
  }
}

// ================= READ IR =================
void readSensors() {
  s1 = (analogRead(IR1) > sensorThreshold) ? 1 : 0;
  s2 = (analogRead(IR2) > sensorThreshold) ? 1 : 0;
  s3 = (analogRead(IR3) > sensorThreshold) ? 1 : 0;
  s4 = (analogRead(IR4) > sensorThreshold) ? 1 : 0;
  s5 = (analogRead(IR5) > sensorThreshold) ? 1 : 0;
}

// ================= ULTRASONIC =================
float getDistance() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);

  long duration = pulseIn(ECHO_PIN, HIGH, 30000);
  if (duration == 0) return 999;

  return duration * 0.034 / 2;
}

// ================= MOTOR =================
void moveForward(int l, int r) {
  leftForward(l);
  rightForward(r);
}

void stopMotors() {
  ledcWrite(CH_LEFT, 0);
  ledcWrite(CH_RIGHT, 0);
}

void leftForward(int s) {
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  ledcWrite(CH_LEFT, s);
}

void leftBackward(int s) {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);
  ledcWrite(CH_LEFT, s);
}

void rightForward(int s) {
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
  ledcWrite(CH_RIGHT, s);
}

void rightBackward(int s) {
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
  ledcWrite(CH_RIGHT, s);
}
