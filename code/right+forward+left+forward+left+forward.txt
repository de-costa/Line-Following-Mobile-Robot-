#include <Arduino.h>
#include <Wire.h>
#include "Adafruit_TCS34725.h"

// ================= MOTOR DRIVER =================
#define ENA 26
#define IN1 27
#define IN2 14
#define ENB 12
#define IN3 13
#define IN4 15

// ================= ULTRASONIC =================
#define TRIG_PIN 4
#define ECHO_PIN 18

// ================= PWM =================
#define CH_LEFT  0
#define CH_RIGHT 1

// ================= COLOR SENSOR =================
Adafruit_TCS34725 tcs = Adafruit_TCS34725(
  TCS34725_INTEGRATIONTIME_50MS,
  TCS34725_GAIN_4X
);

// ================= SPEED =================
int turnSpeed = 150;     // low rotation speed
int forwardSpeed = 180;  // forward speed

// ================= SETUP =================
void setup() {
  Serial.begin(115200);

  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);

  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  ledcSetup(CH_LEFT, 1000, 8);
  ledcSetup(CH_RIGHT, 1000, 8);
  ledcAttachPin(ENA, CH_LEFT);
  ledcAttachPin(ENB, CH_RIGHT);

  if (!tcs.begin()) {
    Serial.println("Color sensor not found");
    while (1);
  }

  stopMotors();
}

// ================= LOOP =================
void loop() {

  float distance = getDistance();
  bool isRed = detectRed();

  if (distance > 0 && distance <= 5.0 && isRed) {
    Serial.println("RED OBJECT → UPDATED LEFT SEQUENCE");

    // 1️⃣ RIGHT ROTATE (1 sec)
    leftForward(turnSpeed);
    rightBackward(turnSpeed);
    delay(1000);

    // 2️⃣ FORWARD (1 sec)
    moveForward(forwardSpeed, forwardSpeed);
    delay(1000);

    // 3️⃣ LEFT ROTATE (0.5 sec)
    leftBackward(turnSpeed);
    rightForward(turnSpeed);
    delay(500);

    // 4️⃣ FORWARD (1 sec)
    moveForward(forwardSpeed, forwardSpeed);
    delay(1000);

    // 5️⃣ LEFT ROTATE (0.5 sec)
    leftBackward(turnSpeed);
    rightForward(turnSpeed);
    delay(500);

    // 6️⃣ FINAL FORWARD (1 sec)
    moveForward(forwardSpeed, forwardSpeed);
    delay(1000);

    stopMotors();
    delay(100);

    return;
  }

  // Default forward
  moveForward(120, 120);
}

// ================= COLOR DETECTION =================
bool detectRed() {
  uint16_t r, g, b, c;
  tcs.getRawData(&r, &g, &b, &c);
  return (r > g && r > b && r > 150);
}

// ================= ULTRASONIC =================
float getDistance() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);

  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);

  long duration = pulseIn(ECHO_PIN, HIGH, 30000);
  return duration * 0.034 / 2;
}

// ================= MOTOR =================
void moveForward(int l, int r) {
  leftForward(l);
  rightForward(r);
}

void stopMotors() {
  ledcWrite(CH_LEFT, 0);
  ledcWrite(CH_RIGHT, 0);
}

void leftForward(int s) {
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  ledcWrite(CH_LEFT, s);
}

void leftBackward(int s) {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);
  ledcWrite(CH_LEFT, s);
}

void rightForward(int s) {
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
  ledcWrite(CH_RIGHT, s);
}

void rightBackward(int s) {
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
  ledcWrite(CH_RIGHT, s);
}
