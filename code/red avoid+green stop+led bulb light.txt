#include <Arduino.h>
#include <Wire.h>
#include "Adafruit_TCS34725.h"

// ================= IR SENSOR PINS =================
#define IR1 34
#define IR2 35
#define IR3 32
#define IR4 33
#define IR5 25

// ================= MOTOR DRIVER =================
#define ENA 26
#define IN1 27
#define IN2 14
#define ENB 12
#define IN3 13
#define IN4 15

// ================= ULTRASONIC =================
#define TRIG_PIN 4
#define ECHO_PIN 18

// ================= LED =================
#define BULB_PIN 2

// ================= PWM =================
#define CH_LEFT  0
#define CH_RIGHT 1

// ================= COLOR SENSOR =================
Adafruit_TCS34725 tcs = Adafruit_TCS34725(
  TCS34725_INTEGRATIONTIME_50MS,
  TCS34725_GAIN_4X
);

// ================= PID =================
float Kp = 28.0;
float Kd = 16.0;
float lastError = 0;

// ================= SPEED =================
int baseSpeed = 150;
int maxSpeed = 200;
int minSpeed = 0;

int sharpTurnSpeed = 220;
int searchSpeed = 140;
int slowForward = 100;

// ================= SENSOR VALUES =================
int s1, s2, s3, s4, s5;
int sensorThreshold = 2000;

// ================= MEMORY =================
int lastDirection = 0;
int sharpTurnMemory = 0;
unsigned long sharpTurnTimeout = 1600;

// ================= SETUP =================
void setup() {
  Serial.begin(115200);

  pinMode(IR1, INPUT);
  pinMode(IR2, INPUT);
  pinMode(IR3, INPUT);
  pinMode(IR4, INPUT);
  pinMode(IR5, INPUT);

  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);

  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  pinMode(BULB_PIN, OUTPUT);

  ledcSetup(CH_LEFT, 1000, 8);
  ledcSetup(CH_RIGHT, 1000, 8);
  ledcAttachPin(ENA, CH_LEFT);
  ledcAttachPin(ENB, CH_RIGHT);

  if (!tcs.begin()) {
    Serial.println("Color sensor not found");
    while (1);
  }
}

// ================= LOOP =================
void loop() {

  float distance = getDistance();
  bool isRed = detectRed();
  bool isGreen = detectGreen();

  // ================= 🟢 GREEN → STOP + BLINK =================
  if (distance > 0 && distance <= 5.0 && isGreen) {
    stopMotors();

    
      digitalWrite(BULB_PIN, HIGH);
      delay(1000);
      digitalWrite(BULB_PIN, LOW);
      delay(1000);
    
    return;
  }

  // ================= 🔴 RED → AVOID =================
  if (distance > 0 && distance <= 5.0 && isRed) {

    leftBackward(150);
    rightBackward(150);
    delay(500);

    leftForward(150);
    rightBackward(150);
    delay(500);

    moveForward(160, 160);
    delay(800);

    leftBackward(150);
    rightForward(150);
    delay(500);

    moveForward(160, 160);
    delay(1000);

    leftBackward(150);
    rightForward(150);
    delay(500);

    unsigned long start = millis();

    while (true) {
      readSensors();

      leftForward(160);
      rightForward(120);

      if (s3 == 0 || s2 == 0 || s4 == 0) break;
      if (millis() - start > 3000) break;
    }

    return;
  }

  // ================= LINE FOLLOW =================
  readSensors();

  // SHARP TURN
  if (s1 == 0 && s5 == 1) { sharpTurnMemory = -1; executeSharpTurn(); return; }
  if (s1 == 1 && s5 == 0) { sharpTurnMemory = 1; executeSharpTurn(); return; }

  if (s1 == 0 && s5 == 0) {
    if (lastDirection < 0) sharpTurnMemory = -1;
    else if (lastDirection > 0) sharpTurnMemory = 1;

    if (sharpTurnMemory != 0) {
      executeSharpTurn();
      return;
    }
  }

  sharpTurnMemory = 0;

  normalPIDTracking();
}

// ================= PID =================
void normalPIDTracking() {

  float error = 0;

  if      (s2 == 1 && s3 == 0 && s4 == 1) { error = 0; lastDirection = 0; }
  else if (s2 == 0 && s3 == 0 && s4 == 1) { error = -1; lastDirection = -1; }
  else if (s2 == 0 && s3 == 1 && s4 == 1) { error = -2; lastDirection = -1; }
  else if (s2 == 1 && s3 == 0 && s4 == 0) { error = 1; lastDirection = 1; }
  else if (s2 == 1 && s3 == 1 && s4 == 0) { error = 2; lastDirection = 1; }
  else if (s2 == 0 && s3 == 0 && s4 == 0) { moveForward(slowForward, slowForward); return; }
  else { recoverLostLine(); return; }

  float derivative = error - lastError;
  float correction = (Kp * error) + (Kd * derivative);
  lastError = error;

  int leftSpeed  = constrain(baseSpeed + correction, minSpeed, maxSpeed);
  int rightSpeed = constrain(baseSpeed - correction, minSpeed, maxSpeed);

  moveForward(leftSpeed, rightSpeed);
}

// ================= SHARP TURN =================
void executeSharpTurn() {
  unsigned long startTime = millis();

  while (true) {
    readSensors();

    if (sharpTurnMemory < 0) {
      leftForward(sharpTurnSpeed);
      rightBackward(sharpTurnSpeed);
    } else {
      leftBackward(sharpTurnSpeed);
      rightForward(sharpTurnSpeed);
    }

    if (s3 == 0 && s1 == 1 && s5 == 1) {
      stopMotors();
      return;
    }

    if (millis() - startTime > sharpTurnTimeout) {
      stopMotors();
      return;
    }
  }
}

// ================= RECOVERY =================
void recoverLostLine() {
  if (lastDirection < 0) {
    leftBackward(searchSpeed);
    rightForward(searchSpeed);
  } else if (lastDirection > 0) {
    leftForward(searchSpeed);
    rightBackward(searchSpeed);
  } else {
    moveForward(90, 90);
  }
}

// ================= COLOR =================
bool detectRed() {
  uint16_t r, g, b, c;
  tcs.getRawData(&r, &g, &b, &c);
  return (r > g && r > b && r > 60);
}

bool detectGreen() {
  uint16_t r, g, b, c;
  tcs.getRawData(&r, &g, &b, &c);
  return (g > r && g > b && g > 60);
}

// ================= ULTRASONIC =================
float getDistance() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);

  long duration = pulseIn(ECHO_PIN, HIGH, 30000);
  if (duration == 0) return 999;

  return duration * 0.034 / 2;
}

// ================= READ IR =================
void readSensors() {
  s1 = (analogRead(IR1) > sensorThreshold) ? 1 : 0;
  s2 = (analogRead(IR2) > sensorThreshold) ? 1 : 0;
  s3 = (analogRead(IR3) > sensorThreshold) ? 1 : 0;
  s4 = (analogRead(IR4) > sensorThreshold) ? 1 : 0;
  s5 = (analogRead(IR5) > sensorThreshold) ? 1 : 0;
}

// ================= MOTOR =================
void moveForward(int l, int r) {
  leftForward(l);
  rightForward(r);
}

void stopMotors() {
  ledcWrite(CH_LEFT, 0);
  ledcWrite(CH_RIGHT, 0);
}

void leftForward(int s) {
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  ledcWrite(CH_LEFT, s);
}

void leftBackward(int s) {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);
  ledcWrite(CH_LEFT, s);
}

void rightForward(int s) {
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
  ledcWrite(CH_RIGHT, s);
}

void rightBackward(int s) {
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
  ledcWrite(CH_RIGHT, s);
}
