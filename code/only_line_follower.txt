#include <Arduino.h>

// ================= IR SENSOR PINS =================
#define IR1 34
#define IR2 35
#define IR3 32
#define IR4 33
#define IR5 25

// ================= MOTOR DRIVER PINS =================
#define ENA 26
#define IN1 27
#define IN2 14

#define ENB 12
#define IN3 13
#define IN4 15

// ================= PWM SETTINGS =================
#define CH_LEFT  0
#define CH_RIGHT 1
#define PWM_FREQ 1000
#define PWM_RES  8

// ================= PID =================
float Kp = 28.0;
float Ki = 0.0;
float Kd = 16.0;

float error = 0, lastError = 0;
float integral = 0, derivative = 0, correction = 0;

// ================= SPEED =================
int baseSpeed = 150;
int maxSpeed = 200;
int minSpeed = 0;

int sharpTurnSpeed = 230;
int searchSpeed = 150;
int slowForward = 100;

// ================= SENSOR VALUES =================
int s1, s2, s3, s4, s5;

// ================= MEMORY =================
int lastDirection = 0;
int sharpTurnMemory = 0;

// ================= TIME =================
unsigned long sharpTurnTimeout = 1600;

// ================= SETUP =================
void setup() {
  Serial.begin(115200);

  pinMode(IR1, INPUT);
  pinMode(IR2, INPUT);
  pinMode(IR3, INPUT);
  pinMode(IR4, INPUT);
  pinMode(IR5, INPUT);

  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  // PWM setup (ESP32)
  ledcSetup(CH_LEFT, PWM_FREQ, PWM_RES);
  ledcSetup(CH_RIGHT, PWM_FREQ, PWM_RES);

  ledcAttachPin(ENA, CH_LEFT);
  ledcAttachPin(ENB, CH_RIGHT);

  stopMotors();
}

// ================= LOOP =================
void loop() {
  readSensors();

  // -------- SHARP TURN --------
  if (s1 == 0 && s5 == 1) {
    sharpTurnMemory = -1;
    executeSharpTurn();
    return;
  }

  if (s1 == 1 && s5 == 0) {
    sharpTurnMemory = 1;
    executeSharpTurn();
    return;
  }

  if (s1 == 0 && s5 == 0) {
    if (lastDirection < 0) sharpTurnMemory = -1;
    else if (lastDirection > 0) sharpTurnMemory = 1;

    if (sharpTurnMemory != 0) {
      executeSharpTurn();
      return;
    }
  }

  sharpTurnMemory = 0;

  // -------- NORMAL PID --------
  normalPIDTracking();
}

// ================= READ SENSORS =================
void readSensors() {
  s1 = digitalRead(IR1);
  s2 = digitalRead(IR2);
  s3 = digitalRead(IR3);
  s4 = digitalRead(IR4);
  s5 = digitalRead(IR5);
}

// ================= PID TRACKING =================
void normalPIDTracking() {

  if (s2 == 1 && s3 == 0 && s4 == 1) {
    error = 0;
    lastDirection = 0;
  }
  else if (s2 == 0 && s3 == 0 && s4 == 1) {
    error = -1;
    lastDirection = -1;
  }
  else if (s2 == 0 && s3 == 1 && s4 == 1) {
    error = -2;
    lastDirection = -1;
  }
  else if (s2 == 1 && s3 == 0 && s4 == 0) {
    error = 1;
    lastDirection = 1;
  }
  else if (s2 == 1 && s3 == 1 && s4 == 0) {
    error = 2;
    lastDirection = 1;
  }
  else if (s2 == 0 && s3 == 0 && s4 == 0) {
    moveForward(slowForward, slowForward);
    return;
  }
  else if (s2 == 1 && s3 == 1 && s4 == 1) {
    recoverLostLine();
    return;
  }
  else {
    recoverLostLine();
    return;
  }

  integral += error;
  integral = constrain(integral, -40, 40);

  derivative = error - lastError;
  correction = Kp * error + Ki * integral + Kd * derivative;
  lastError = error;

  int leftSpeed  = baseSpeed + correction;
  int rightSpeed = baseSpeed - correction;

  leftSpeed  = constrain(leftSpeed, minSpeed, maxSpeed);
  rightSpeed = constrain(rightSpeed, minSpeed, maxSpeed);

  moveForward(leftSpeed, rightSpeed);
}

// ================= SHARP TURN =================
void executeSharpTurn() {
  unsigned long startTime = millis();

  while (true) {
    readSensors();

    if (sharpTurnMemory < 0) {
      leftForward(sharpTurnSpeed);
      rightBackward(sharpTurnSpeed);
    } else if (sharpTurnMemory > 0) {
      leftBackward(sharpTurnSpeed);
      rightForward(sharpTurnSpeed);
    }

    if (s3 == 0 && s1 == 1 && s5 == 1) {
      stopMotors();
      sharpTurnMemory = 0;
      return;
    }

    if (millis() - startTime > sharpTurnTimeout) {
      stopMotors();
      sharpTurnMemory = 0;
      return;
    }
  }
}

// ================= LOST LINE =================
void recoverLostLine() {
  if (lastDirection < 0) {
    leftBackward(searchSpeed);
    rightForward(searchSpeed);
  }
  else if (lastDirection > 0) {
    leftForward(searchSpeed);
    rightBackward(searchSpeed);
  }
  else {
    moveForward(90, 90);
  }
}

// ================= MOTOR CONTROL =================
void moveForward(int leftSpeed, int rightSpeed) {
  leftForward(leftSpeed);
  rightForward(rightSpeed);
}

void stopMotors() {
  ledcWrite(CH_LEFT, 0);
  ledcWrite(CH_RIGHT, 0);

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}

void leftForward(int speedVal) {
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  ledcWrite(CH_LEFT, speedVal);
}

void leftBackward(int speedVal) {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);
  ledcWrite(CH_LEFT, speedVal);
}

void rightForward(int speedVal) {
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
  ledcWrite(CH_RIGHT, speedVal);
}

void rightBackward(int speedVal) {
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
  ledcWrite(CH_RIGHT, speedVal);
}
