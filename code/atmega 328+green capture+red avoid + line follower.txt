#include <Arduino.h>
#include <Wire.h>
#include "Adafruit_TCS34725.h"

// ================= IR SENSOR PINS =================
#define IR1 34
#define IR2 35
#define IR3 32
#define IR4 33
#define IR5 25

// ================= MOTOR DRIVER =================
#define ENA 26
#define IN1 27
#define IN2 14
#define ENB 12
#define IN3 13
#define IN4 15

// ================= ULTRASONIC =================
#define TRIG_PIN 4
#define ECHO_PIN 18

// ================= BULB =================
#define BULB_PIN 2

// ================= PWM =================
#define CH_LEFT  0
#define CH_RIGHT 1

// ================= COLOR SENSOR =================
Adafruit_TCS34725 tcs = Adafruit_TCS34725(
  TCS34725_INTEGRATIONTIME_50MS,
  TCS34725_GAIN_4X
);

// ================= PID =================
float Kp = 28.0;

// ================= SPEED =================
int baseSpeed = 150;
int maxSpeed = 200;
int minSpeed = 0;

int sharpTurnSpeed = 230;
int searchSpeed = 150;
int slowForward = 100;

// ================= SENSOR VALUES =================
int s1, s2, s3, s4, s5;

// ================= MEMORY =================
int lastDirection = 0;
int sharpTurnMemory = 0;
unsigned long sharpTurnTimeout = 1600;

// ================= SETUP =================
void setup() {
  Serial.begin(115200);

  pinMode(IR1, INPUT);
  pinMode(IR2, INPUT);
  pinMode(IR3, INPUT);
  pinMode(IR4, INPUT);
  pinMode(IR5, INPUT);

  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);

  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  pinMode(BULB_PIN, OUTPUT);
  digitalWrite(BULB_PIN, LOW);

  ledcSetup(CH_LEFT, 1000, 8);
  ledcSetup(CH_RIGHT, 1000, 8);
  ledcAttachPin(ENA, CH_LEFT);
  ledcAttachPin(ENB, CH_RIGHT);

  if (!tcs.begin()) {
    Serial.println("Color sensor not found");
    while (1);
  }

  stopMotors();
}

// ================= LOOP =================
void loop() {

  float distance = getDistance();
  bool isRed = detectRed();
  bool isGreen = detectGreen();

  Serial.print("Distance: ");
  Serial.print(distance);
  Serial.print(" RED: ");
  Serial.print(isRed);
  Serial.print(" GREEN: ");
  Serial.println(isGreen);

  // ================= OBJECT DETECT =================
  if (distance > 0 && distance <= 5.0) {

    // ===== RED OBJECT =====
    if (isRed) {
      stopMotors();
      delay(1000);

      leftForward(150);
      rightBackward(150);
      delay(500);

      moveForward(160, 160);
      delay(800);

      leftBackward(150);
      rightForward(150);
      delay(500);

      moveForward(160, 160);
      delay(1000);

      leftBackward(150);
      rightForward(150);
      delay(500);

      return;
    }

    // ===== GREEN OBJECT =====
    if (isGreen) {
      stopMotors();
      delay(1000);

      digitalWrite(BULB_PIN, HIGH);
      delay(2000);

      return;
    }

  } else {
    // No object → bulb OFF
    digitalWrite(BULB_PIN, LOW);
  }

  // ================= LINE FOLLOW =================
  readSensors();

  if (s1 == 0 && s5 == 1) { sharpTurnMemory = -1; executeSharpTurn(); return; }
  if (s1 == 1 && s5 == 0) { sharpTurnMemory = 1; executeSharpTurn(); return; }

  if (s1 == 0 && s5 == 0) {
    if (lastDirection < 0) sharpTurnMemory = -1;
    else if (lastDirection > 0) sharpTurnMemory = 1;

    if (sharpTurnMemory != 0) {
      executeSharpTurn();
      return;
    }
  }

  sharpTurnMemory = 0;
  normalPIDTracking();
}

// ================= COLOR =================
bool detectRed() {
  uint16_t r, g, b, c;
  tcs.getRawData(&r, &g, &b, &c);

  if (c == 0) return false;

  float rr = (float)r / c;
  float gr = (float)g / c;
  float br = (float)b / c;

  return (rr > 0.45 && rr > gr && rr > br);
}

bool detectGreen() {
  uint16_t r, g, b, c;
  tcs.getRawData(&r, &g, &b, &c);

  if (c == 0) return false;

  float gr = (float)g / c;
  float rr = (float)r / c;
  float br = (float)b / c;

  return (gr > 0.35 && gr > rr && gr > br);
}

// ================= ULTRASONIC =================
float getDistance() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);

  long duration = pulseIn(ECHO_PIN, HIGH, 30000);
  return duration * 0.034 / 2;
}

// ================= READ IR =================
void readSensors() {
  s1 = digitalRead(IR1);
  s2 = digitalRead(IR2);
  s3 = digitalRead(IR3);
  s4 = digitalRead(IR4);
  s5 = digitalRead(IR5);
}

// ================= PID =================
void normalPIDTracking() {

  float error = 0;

  if (s2 == 1 && s3 == 0 && s4 == 1) { error = 0; lastDirection = 0; }
  else if (s2 == 0 && s3 == 0 && s4 == 1) { error = -1; lastDirection = -1; }
  else if (s2 == 0 && s3 == 1 && s4 == 1) { error = -2; lastDirection = -1; }
  else if (s2 == 1 && s3 == 0 && s4 == 0) { error = 1; lastDirection = 1; }
  else if (s2 == 1 && s3 == 1 && s4 == 0) { error = 2; lastDirection = 1; }
  else if (s2 == 0 && s3 == 0 && s4 == 0) { moveForward(slowForward, slowForward); return; }
  else { recoverLostLine(); return; }

  int leftSpeed  = constrain(baseSpeed + (Kp * error), minSpeed, maxSpeed);
  int rightSpeed = constrain(baseSpeed - (Kp * error), minSpeed, maxSpeed);

  moveForward(leftSpeed, rightSpeed);
}

// ================= SHARP TURN =================
void executeSharpTurn() {
  unsigned long startTime = millis();

  while (true) {
    readSensors();

    if (sharpTurnMemory < 0) {
      leftForward(sharpTurnSpeed);
      rightBackward(sharpTurnSpeed);
    } else {
      leftBackward(sharpTurnSpeed);
      rightForward(sharpTurnSpeed);
    }

    if (s3 == 0 && s1 == 1 && s5 == 1) return;
    if (millis() - startTime > sharpTurnTimeout) return;
  }
}

// ================= RECOVERY =================
void recoverLostLine() {
  if (lastDirection < 0) {
    leftBackward(searchSpeed);
    rightForward(searchSpeed);
  }
  else if (lastDirection > 0) {
    leftForward(searchSpeed);
    rightBackward(searchSpeed);
  }
  else {
    moveForward(90, 90);
  }
}

// ================= MOTOR =================
void moveForward(int l, int r) {
  leftForward(l);
  rightForward(r);
}

void stopMotors() {
  ledcWrite(CH_LEFT, 0);
  ledcWrite(CH_RIGHT, 0);
}

void leftForward(int s) {
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  ledcWrite(CH_LEFT, s);
}

void leftBackward(int s) {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);
  ledcWrite(CH_LEFT, s);
}

void rightForward(int s) {
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
  ledcWrite(CH_RIGHT, s);
}

void rightBackward(int s) {
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
  ledcWrite(CH_RIGHT, s);
}
