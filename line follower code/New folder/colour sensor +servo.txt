#include <Wire.h>
#include <ESP32Servo.h>
#include "Adafruit_TCS34725.h"

// ================= SERVO =================
Servo grip;
Servo hang;

#define GRIP_PIN 23
#define HANG_PIN 19

// ================= COLOR SENSOR =================
Adafruit_TCS34725 tcs = Adafruit_TCS34725(
  TCS34725_INTEGRATIONTIME_50MS,
  TCS34725_GAIN_4X
);

void setup() {
  Serial.begin(115200);

  grip.attach(GRIP_PIN);
  hang.attach(HANG_PIN);

  // 🔼 Initial position
  grip.write(170);   // open
  hang.write(100);   // start at 100

  if (!tcs.begin()) {
    Serial.println("Color sensor not found");
    while (1);
  }
}

void loop() {

  uint16_t r, g, b, c;
  tcs.getRawData(&r, &g, &b, &c);

  Serial.print("R: "); Serial.print(r);
  Serial.print(" G: "); Serial.print(g);
  Serial.print(" B: "); Serial.println(b);

  // 🟢 GREEN DETECTED
  if (g > r && g > b && g > 50) {
    Serial.println("GREEN DETECTED!");

    // Step 1: hang goes UP (100 → 170)
    hang.write(170);
    delay(500);

    // Step 2: grip closes
    grip.write(120);
    delay(1000);

    // Step 3: hold position (like lift stage)
    hang.write(100);
    delay(10000);

    // Step 4: release grip
    grip.write(170);
    delay(100);

    // Step 5: reset hang back to 100
    hang.write(100);
    delay(100);
  }
  else {
    // Idle
    grip.write(170);
    hang.write(100);
  }

  delay(200);
}
